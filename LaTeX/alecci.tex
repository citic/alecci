% Alecci language definition for the listings package.
% Required in preamble: \usepackage{listings} and \usepackage{xcolor}
%
% Keyword classes:
%   [1]  control-flow, structure, logical operators, I/O
%   [2]  primitive scalar types
%   [3]  concurrency-primitive type keywords (also used as constructors)
%   [4]  built-in procedures and functions
%
% Usage:
%   \begin{lstlisting}[language=Alecci]
%   ...code...
%   \end{lstlisting}
%
% Or for a default style applied to all Alecci listings:
%   \lstset{language=Alecci, basicstyle=\ttfamily\small, ...}

\lstdefinelanguage{Alecci}{%
  %
  % -------------------------------------------------------------------------
  % Class 1: control-flow, declarations, structure, logical/bitwise operators
  % -------------------------------------------------------------------------
  keywords=[1]{%
    procedure, function, return, end,%
    if, then, else,%
    while, do,%
    for, parallel,%
    break, case,%
    shared, mutable, const, atomic,%
    as, of, to, pointed, by,%
    record, enum, pointer,%
    with,%
    variant,%
    and, or, not, xor, mod,%
    true, false,%
    print, scan%
  },%
  keywordstyle=[1]{\bfseries\color{SteelBlue}},%
  %
  % -------------------------------------------------------------------------
  % Class 2: primitive scalar types
  % -------------------------------------------------------------------------
  keywords=[2]{%
    int, float, bool, char, string,%
    i8, i16, i32, i64,%
    u8, u16, u32, u64,%
    f32, f64, f128%
  },%
  keywordstyle=[2]{\color{CadetBlue}},%
  %
  % -------------------------------------------------------------------------
  % Class 3: concurrency-primitive type keywords / constructors
  % -------------------------------------------------------------------------
  keywords=[3]{%
    thread, semaphore, mutex, barrier, condvar, queue, array%
  },%
  keywordstyle=[3]{\color{MediumOrchid}},%
  %
  % -------------------------------------------------------------------------
  % Class 4: built-in procedures and functions
  % -------------------------------------------------------------------------
  keywords=[4]{%
    create_threads, create_thread,%
    join_threads, join_thread,%
    lock, unlock,%
    wait, signal, barrier_wait,%
    cond_wait, cond_signal, cond_broadcast,%
    sleep, rand, seed,%
    enqueue, dequeue, queue_size, queue_capacity,%
    variant_type, variant_copy, variant_equals, variant_to_string%
  },%
  keywordstyle=[4]{\color{Peru}},%
  %
  % -------------------------------------------------------------------------
  % Comments: // to end of line
  % -------------------------------------------------------------------------
  comment=[l]{//},%
  commentstyle={\itshape\color{DimGray}},%
  %
  % -------------------------------------------------------------------------
  % String literals: "..." and backtick `...` (formatted strings)
  % -------------------------------------------------------------------------
  string=[b]{"},%
  morestring=[b]{`},%
  stringstyle={\color{FireBrick}},%
  %
  % -------------------------------------------------------------------------
  % Multi-character operators: prevent listings from breaking them apart.
  % If mathtools is available, replace := with the proper \coloneqq symbol by
  % adding  literate+={{:=}{{\ensuremath{\coloneqq}}}2}  after \input-ing this
  % file. The := and != entries keep those operators as single tokens.
  % NOTE: do NOT add self-referential rules for < > operators (<< >> <= >=).
  % In a babel `spanish` document < and > are active (guillemet shorthand),
  % so re-emitting them via literate triggers infinite expansion / TeX
  % capacity overflow. The raw characters render fine straight from listings.
  % The accented-character entries let UTF-8 Spanish text (e.g. comments and
  % the operator glossary) appear inside listings without an inputenc error.
  % -------------------------------------------------------------------------
  literate=%
    {:=}{{:=}}2%
    {!=}{{!=}}2%
    {á}{{\'a}}1 {é}{{\'e}}1 {í}{{\'i}}1 {ó}{{\'o}}1 {ú}{{\'u}}1%
    {Á}{{\'A}}1 {É}{{\'E}}1 {Í}{{\'I}}1 {Ó}{{\'O}}1 {Ú}{{\'U}}1%
    {ñ}{{\~n}}1 {Ñ}{{\~N}}1 {ü}{{\"u}}1 {Ü}{{\"U}}1,%
  %
  % -------------------------------------------------------------------------
  % Miscellaneous
  % -------------------------------------------------------------------------
  extendedchars=true,%    % allow bytes > 127 (UTF-8 accented characters)
  sensitive=true,%        % Alecci is case-sensitive
  morecomment=[l]{//},%   % ensure // is always a line comment
  breaklines=true,%       % wrap long lines instead of overflowing the cell/page
  breakatwhitespace=true,%% prefer breaking at spaces
  breakindent=0pt,%
  columns=flexible,%
  keepspaces=true%
}

% ---------------------------------------------------------------------------
% AlecciPython — Python pseudocode style matching the Alecci colour palette.
% Used for algorithm listings that present Alecci analyses in Python notation.
%
% Keyword classes mirror the Alecci and AlecciC definitions:
%   [1]  control-flow, declarations, logical operators
%   [2]  built-in types and functions
%   [3]  concurrency / Alecci domain concepts
%   [4]  algorithm-specific procedures
% ---------------------------------------------------------------------------
\lstdefinelanguage{AlecciPython}{%
  %
  % -------------------------------------------------------------------------
  % Class 1: control flow, declarations, logical operators
  % -------------------------------------------------------------------------
  keywords=[1]{%
    def, return, yield,%
    for, in, while, break, continue, pass,%
    if, elif, else,%
    and, or, not, is,%
    with, as, from, import, class,%
    lambda, del, raise, try, except, finally%
  },%
  keywordstyle=[1]{\bfseries\color{SteelBlue}},%
  %
  % -------------------------------------------------------------------------
  % Class 2: built-in types and functions
  % -------------------------------------------------------------------------
  keywords=[2]{%
    set, dict, list, tuple, str, int, float, bool,%
    None, True, False,%
    sum, abs, max, min, len, range, print,%
    frozenset, enumerate, zip, sorted, reversed,%
    isinstance, hasattr, getattr, setattr%
  },%
  keywordstyle=[2]{\color{CadetBlue}},%
  %
  % -------------------------------------------------------------------------
  % Class 3: concurrency and Alecci domain concepts
  % -------------------------------------------------------------------------
  keywords=[3]{%
    wait, signal, semaphore, procedure, function,%
    create_thread, create_threads,%
    WHITE, GRAY, BLACK%
  },%
  keywordstyle=[3]{\color{MediumOrchid}},%
  %
  % -------------------------------------------------------------------------
  % Class 4: algorithm-specific procedures (domain names)
  % -------------------------------------------------------------------------
  keywords=[4]{%
    analyze,%
    collect_symbols, compute_delta, check_depletion,%
    find_recursive_cycles,%
    walk, warn, calls, dfs%
  },%
  keywordstyle=[4]{\color{Peru}},%
  %
  % -------------------------------------------------------------------------
  % Comments: # to end of line
  % -------------------------------------------------------------------------
  comment=[l]{\#},%
  commentstyle={\itshape\color{DimGray}},%
  %
  % -------------------------------------------------------------------------
  % String literals: "..." and '...'
  % -------------------------------------------------------------------------
  string=[b]{"},%
  morestring=[b]{'},%
  stringstyle={\color{FireBrick}},%
  %
  % -------------------------------------------------------------------------
  % Operators and accented characters
  % -------------------------------------------------------------------------
  literate=%
    {->}{{->}}2 {==}{{==}}2 {!=}{{!=}}2%
    {+=}{{+=}}2 {-=}{{-=}}2 {**}{{**}}2%
    {á}{{\'a}}1 {é}{{\'e}}1 {í}{{\'i}}1 {ó}{{\'o}}1 {ú}{{\'u}}1%
    {Á}{{\'A}}1 {É}{{\'E}}1 {Í}{{\'I}}1 {Ó}{{\'O}}1 {Ú}{{\'U}}1%
    {ñ}{{\~n}}1 {Ñ}{{\~N}}1 {ü}{{\"u}}1 {Ü}{{\"U}}1,%
  %
  % -------------------------------------------------------------------------
  % Miscellaneous
  % -------------------------------------------------------------------------
  extendedchars=true,%
  sensitive=true,%
  breaklines=true,%
  breakatwhitespace=true,%
  breakindent=0pt,%
  columns=flexible,%
  keepspaces=true%
}
